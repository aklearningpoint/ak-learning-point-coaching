// ========================================== // STUDENT PORTAL ENGINE - ASSENT PUBLIC SCHOOL // ========================================== var currentStudent = null; // Smart Class Matcher: Handles "5", "Class 5", "Nur & Play", etc. function normalizeClass(cls) { return String(cls || "").toLowerCase().replace(/[^a-z0-9]/g, ""); } function isClassEqual(classA, classB) { var a = normalizeClass(classA); var b = normalizeClass(classB); if (!a || !b) return false; return a === b || a === ("class" + b) || ("class" + a) === b; } // 1. GATEWAY: STUDENT VERIFICATION & LOGIN function verifyStudentCandidate() { var selectedClass = document.getElementById("loginClass").value.trim(); var enteredRoll = parseInt(document.getElementById("loginRoll").value, 10); if (isNaN(enteredRoll)) { alert("Kripya valid Roll Number enter karein!"); return; } // Fetch students from school-db.js cache or cloud var studentsList = []; if (typeof schoolDB !== "undefined" && typeof schoolDB.getStudents === "function") { studentsList = schoolDB.getStudents(); } else { try { studentsList = JSON.parse(localStorage.getItem("ak_school_students") || "[]"); } catch(e) { studentsList = []; } } if (!studentsList || studentsList.length === 0) { alert("Students data abhi load ho raha hai ya sheet se fetch nahi hua. Kripya 2 second baad dobara prayas karein!"); return; } // Find candidate by Class and Roll Number var candidate = studentsList.find(function(s) { var rollMatch = parseInt(s.rollNo || s.roll, 10) === enteredRoll; var classMatch = isClassEqual(s.className || s.class, selectedClass); return rollMatch && classMatch; }); if (!candidate) { alert("Student record nahi mila!\nClass: " + selectedClass + " | Roll No: " + enteredRoll + "\nKripya Class aur Roll Number check karein."); return; } currentStudent = candidate; // Transition UI to Workspace document.getElementById("studentAuthOverlay").style.display = "none"; document.getElementById("studentWorkspace").style.display = "block"; // Populate Student Details document.getElementById("welcomeBanner").innerText = "Welcome, " + (candidate.name || "Student"); populateStudentProfile(candidate); renderStudentIDCard(candidate); renderStudentResults(candidate); } // 2. LOGOUT / EXIT function logoutStudent() { currentStudent = null; document.getElementById("studentWorkspace").style.display = "none"; document.getElementById("studentAuthOverlay").style.display = "flex"; document.getElementById("loginRoll").value = ""; } // 3. TAB NAVIGATION function switchTab(tabId) { var panes = document.querySelectorAll(".tab-pane"); panes.forEach(function(pane) { pane.classList.remove("active"); }); var btns = document.querySelectorAll(".tab-btn"); btns.forEach(function(btn) { btn.classList.remove("active"); }); var targetPane = document.getElementById(tabId); if (targetPane) targetPane.classList.add("active"); if (event && event.currentTarget) { event.currentTarget.classList.add("active"); } } // 4. POPULATE PROFILE TAB function populateStudentProfile(s) { document.getElementById("stuRegNo").value = s.regNo || ""; document.getElementById("stuFather").value = s.father || ""; document.getElementById("stuMother").value = s.mother || ""; document.getElementById("stuGender").value = s.gender || "Male"; document.getElementById("stuTransport").value = s.transport || "School Transport"; document.getElementById("stuMob1").value = s.mobile1 || s.mobile || ""; document.getElementById("stuMob2").value = s.mobile2 || ""; document.getElementById("stuDOB").value = s.dob || ""; } // 5. SAVE PROFILE SELF-UPDATE function saveStudentSelfDetails(e) { e.preventDefault(); if (!currentStudent) return; currentStudent.regNo = document.getElementById("stuRegNo").value.trim(); currentStudent.father = document.getElementById("stuFather").value.trim(); currentStudent.mother = document.getElementById("stuMother").value.trim(); currentStudent.gender = document.getElementById("stuGender").value; currentStudent.transport = document.getElementById("stuTransport").value; currentStudent.mobile1 = document.getElementById("stuMob1").value.trim(); currentStudent.mobile2 = document.getElementById("stuMob2").value.trim(); currentStudent.dob = document.getElementById("stuDOB").value; // Save to Local DB if (typeof schoolDB !== "undefined" && typeof schoolDB.updateStudent === "function") { schoolDB.updateStudent(currentStudent); } // Sync to Cloud Sheet if API is present if (typeof CLOUD_SYNC_URL !== "undefined" && CLOUD_SYNC_URL) { try { fetch(CLOUD_SYNC_URL, { method: "POST", mode: "no-cors", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ action: "updateStudentDetails", student: currentStudent }) }); } catch(err) { console.warn("Cloud sync error:", err); } } alert("Profile information successfully update ho gayi!"); renderStudentIDCard(currentStudent); } // 6.
